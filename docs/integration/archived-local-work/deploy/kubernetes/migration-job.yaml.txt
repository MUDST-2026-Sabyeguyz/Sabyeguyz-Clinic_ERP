apiVersion: batch/v1
kind: Job
metadata:
  name: __MIGRATION_JOB__
spec:
  backoffLimit: 0
  activeDeadlineSeconds: 300
  ttlSecondsAfterFinished: 86400
  template:
    metadata:
      labels:
        app: clinic-migrate
    spec:
      restartPolicy: Never
      automountServiceAccountToken: false
      containers:
        - name: migrate
          image: __BACKEND_IMAGE__
          imagePullPolicy: IfNotPresent
          command:
            - java
            - -Dloader.main=com.sabyeguyz.clinicv2.config.DatabaseMigrationMain
            - -cp
            - /app/app.jar
            - org.springframework.boot.loader.launch.PropertiesLauncher
          env:
            - {name: SPRING_DATASOURCE_URL, value: 'jdbc:postgresql://postgres:5432/__DATABASE_NAME__'}
            - {name: SPRING_DATASOURCE_USERNAME, value: clinic}
            - {name: CLINIC_SEED_MODE, value: '__SEED_MODE__'}
            - name: SPRING_DATASOURCE_PASSWORD
              valueFrom:
                secretKeyRef: {name: clinic-secrets, key: SPRING_DATASOURCE_PASSWORD}
