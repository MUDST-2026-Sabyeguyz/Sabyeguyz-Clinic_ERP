package com.sabyeguyz.clinicv2.config;

import org.flywaydb.core.Flyway;

/** Kubernetes migration job entry point, using exactly the SQL files packaged in the deployed backend image. */
public final class DatabaseMigrationMain {
    private DatabaseMigrationMain() {}

    public static void main(String[] args) {
        String url = required("SPRING_DATASOURCE_URL");
        String user = required("SPRING_DATASOURCE_USERNAME");
        String password = required("SPRING_DATASOURCE_PASSWORD");
        var configuration = Flyway.configure().dataSource(url, user, password)
                .connectRetries(12).cleanDisabled(true);
        DatabaseMigrations.migrate(configuration, System.getenv().getOrDefault("CLINIC_SEED_MODE", "auto"));
    }

    private static String required(String name) {
        String value = System.getenv(name);
        if (value == null || value.isBlank()) throw new IllegalArgumentException("Missing " + name);
        return value;
    }
}
